# Apache .htaccess file for www.immobilienghumman.de
# Redirects for specific pages

# Enable RewriteEngine
RewriteEngine On

# Disable MultiViews to prevent Apache content negotiation conflicts
Options -MultiViews

# ==================================================================
# SITEMAP – dynamische Generierung via sitemap.php
# ==================================================================
RewriteCond %{REQUEST_FILENAME} !-f
RewriteRule ^sitemap\.xml$ sitemap.php [L]

# ==================================================================
# CUSTOM 404 ERROR PAGE
# ==================================================================
ErrorDocument 404 /404.php

# ==================================================================
# CUSTOM 500 ERROR PAGE
# ==================================================================
ErrorDocument 500 /500.php

# ==================================================================
# HTTPS ERZWINGEN
# ==================================================================
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# Canonical Host auf www.immobilienghumman.de
RewriteCond %{HTTP_HOST} !^www\.immobilienghumman\.de$ [NC]
RewriteRule ^(.*)$ https://www.immobilienghumman.de/$1 [R=301,L]

# Alte Domainvarianten auf neue Canonical-Domain umleiten
RewriteCond %{HTTP_HOST} ^(www\.)?immobilien-ghumman\.de$ [NC,OR]
RewriteCond %{HTTP_HOST} ^immobilienghumman\.de$ [NC]
RewriteRule ^(.*)$ https://www.immobilienghumman.de/$1 [R=301,L]

# Redirect https://www.immobilienghumman.de/finden-sie-uns/ to homepage
RewriteCond %{HTTP_HOST} ^www\.immobilienghumman\.de$ [NC]
RewriteRule ^finden-sie-uns/?$ https://www.immobilienghumman.de/ [R=301,L]

# Alte Property-Slug-Query auf kanonische Slug-Route umleiten
RewriteCond %{REQUEST_URI} ^/property-detail/?$ [NC]
RewriteCond %{QUERY_STRING} (^|&)slug=([^&]+) [NC]
RewriteRule ^property-detail/?$ /immobilie/%2? [R=301,L,NE]

# Clean Property Route: /immobilie/<slug>
RewriteRule ^immobilie/([^/]+)/?$ property-detail.php?slug=$1 [L,QSA]

# Ratgeber-Artikel: /ratgeber/<slug> -> ratgeber-artikel.php
RewriteRule ^ratgeber/([a-z0-9-]+)/?$ ratgeber-artikel.php?slug=$1 [L,QSA]

# Hausverwaltung-Unterseiten auf die zentrale Accordion-Seite umleiten
RewriteRule ^hausverwaltung-mietverwaltung/?$ /hausverwaltung#leistung-mietverwaltung [R=301,L,NE]
RewriteRule ^hausverwaltung-nebenkostenabrechnung/?$ /hausverwaltung#leistung-nebenkostenabrechnung [R=301,L,NE]
RewriteRule ^hausverwaltung-instandhaltung/?$ /hausverwaltung#leistung-instandhaltung [R=301,L,NE]
RewriteRule ^hausverwaltung-eigentuemerversammlung/?$ /hausverwaltung#leistung-eigentuemerversammlung [R=301,L,NE]
RewriteRule ^hausverwaltung-digital/?$ /hausverwaltung#leistung-digital [R=301,L,NE]
RewriteRule ^hausverwaltung-rechtliches/?$ /hausverwaltung#leistung-rechtliches [R=301,L,NE]

# Security Headers (recommended)
<IfModule mod_headers.c>
    Header always set X-Frame-Options "SAMEORIGIN"
    Header always set X-Content-Type-Options "nosniff"
    Header always set X-XSS-Protection "1; mode=block"
    Header always set Strict-Transport-Security "max-age=63072000; includeSubDomains; preload"
    Header always set Referrer-Policy "strict-origin-when-cross-origin"
    Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=()"
    # Content Security Policy
    Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' https://www.googletagmanager.com https://www.google-analytics.com https://cdnjs.cloudflare.com https://unpkg.com https://challenges.cloudflare.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com https://cdnjs.cloudflare.com https://unpkg.com; font-src 'self' https://fonts.gstatic.com https://cdnjs.cloudflare.com; img-src 'self' data: https://res.cloudinary.com https://images.unsplash.com https://images.pexels.com https://www.google-analytics.com https://lh3.googleusercontent.com https://*.tile.openstreetmap.org; media-src 'self' https://res.cloudinary.com; connect-src 'self' https://www.google-analytics.com https://nominatim.openstreetmap.org https://challenges.cloudflare.com; frame-src 'self' https://challenges.cloudflare.com; object-src 'none'; base-uri 'self'; form-action 'self';"
    # Remove sensitive headers
    Header always unset X-Powered-By
    Header always unset Server
</IfModule>

# Performance Optimization - Enable Compression
<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/plain
    AddOutputFilterByType DEFLATE text/html
    AddOutputFilterByType DEFLATE text/xml
    AddOutputFilterByType DEFLATE text/css
    AddOutputFilterByType DEFLATE application/xml
    AddOutputFilterByType DEFLATE application/xhtml+xml
    AddOutputFilterByType DEFLATE application/rss+xml
    AddOutputFilterByType DEFLATE application/javascript
    AddOutputFilterByType DEFLATE application/x-javascript
</IfModule>

# Browser Caching
<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresByType image/jpg "access plus 1 month"
    ExpiresByType image/jpeg "access plus 1 month"
    ExpiresByType image/gif "access plus 1 month"
    ExpiresByType image/png "access plus 1 month"
    ExpiresByType image/webp "access plus 1 month"
    ExpiresByType image/svg+xml "access plus 1 month"
    ExpiresByType text/css "access plus 1 week"
    ExpiresByType application/pdf "access plus 1 month"
    # JavaScript: Kürzere Cache-Zeit für häufige Updates
    ExpiresByType text/javascript "access plus 1 hour"
    ExpiresByType application/javascript "access plus 1 hour"
    ExpiresByType font/woff2 "access plus 1 year"
    ExpiresByType font/woff "access plus 1 year"
    ExpiresByType application/font-woff2 "access plus 1 year"
    ExpiresByType text/html "access plus 0 seconds"
    ExpiresByType application/xhtml+xml "access plus 0 seconds"
</IfModule>

# Cache-Control Header für dynamische Inhalte
<IfModule mod_headers.c>
    <FilesMatch "\.(js)$">
        Header set Cache-Control "max-age=3600, must-revalidate"
    </FilesMatch>
    <FilesMatch "\.(html|htm|php)$">
        Header set Cache-Control "no-cache, no-store, must-revalidate"
        Header set Pragma "no-cache"
        Header set Expires 0
    </FilesMatch>
</IfModule>

# ==================================================================
# CLEAN URLS (ohne .php/.html/.asp)
# ==================================================================
# 1) Alte URL mit Endung auf Clean URL umleiten (nur Public-Bereich)
RewriteCond %{THE_REQUEST} \s/+([^\s?]+)\.(php|html|asp)(?:[\s?]|$) [NC]
RewriteCond %1 !(^|/)(admin|api|includes|tools|database|uploads)(?:/|$) [NC]
RewriteRule ^ /%1 [R=301,L,NE]

# 2) Einheitliche URL-Form ohne trailing slash (außer echte Ordner)
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule ^(.+)/$ /$1 [R=301,L]

# 3) Clean URL intern auf .php auflösen, wenn Datei existiert (Public)
RewriteCond %{REQUEST_URI} !^/(admin|api|includes|tools|database|uploads)(?:/|$) [NC]
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_FILENAME}.php -f
RewriteRule ^(.+)$ $1.php [L,QSA]

# ==================================================================
# VERZEICHNIS-BROWSING DEAKTIVIEREN
# ==================================================================
Options -Indexes

# ==================================================================
# SENSIBLE DATEIEN SCHÜTZEN
# ==================================================================
<FilesMatch "^\.">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
</FilesMatch>

<FilesMatch "\.(env|config|yml|yaml|log|sql|bak)$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
</FilesMatch>

# PHP-Konfigurationsdateien per HTTP-Zugriff schützen
<FilesMatch "config\.local\.php$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
</FilesMatch>

# Datenbank-Schema und Seed-Dateien schützen
<FilesMatch "\.(sql)$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
</FilesMatch>

# includes/ und tools/ von direktem Aufruf schützen
<IfModule mod_rewrite.c>
    RewriteRule ^includes/  - [F,L]
    RewriteRule ^tools/     - [F,L]
    RewriteRule ^database/(?!init\.php) - [F,L]
</IfModule>
